  template = template.replace('{{EVENT_DATE}}', formData.eventDate);
  template = template.replace('{{EVENT_ID}}', formData.eventID);
  // ... replace all other placeholders

    to: formData.contactEmail,
    subject: 'Thank You for Your Event Inquiry - Event ID: ' + formData.eventID,
    body: template
  });
